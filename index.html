import { serve } from "https://deno.land/std@0.168.0/http/server.ts"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", }; serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } try { const { question, snapshot } = await req.json(); if (!question) { return new Response(JSON.stringify({ error: "no question provided" }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } const apiKey = Deno.env.get("ANTHROPIC_API_KEY"); if (!apiKey) { return new Response(JSON.stringify({ error: "API key not configured" }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } const systemPrompt = `You are a friendly, warm financial assistant for Mary and Garrett Jones, a married couple managing their household finances together through an app called Float. Your job is to give them honest, clear, encouraging answers about their money situation. Be direct but kind — they're working hard to get on top of their finances and making real progress. Key context about their situation: - They have a Best Buy deferred interest balance of $2,306 due by Oct 23, 2026 — this is a priority. Missing it means a big interest hit. - Garrett works at CFA (~$1,150/month). Mary does freelance design work (variable income). - They have multiple accounts: Bills *0304, Joint *9570, MMJ Freelance *4146. - They're tracking savings goals for Christmas ($1,000), Annabelle's birthday ($300), and JJ's birthday ($300). - They've made some financial mistakes in the past and are actively working to improve — be encouraging, not judgmental. When answering: - Lead with a direct yes/no or one clear sentence - Follow with ONE reason or one relevant number — that's it - If something urgent needs flagging, add it in a second sentence max - Think text message, not report. Short, warm, done.`; const userMessage = `Here is our current financial snapshot: ${JSON.stringify(snapshot, null, 2)} Our question: ${question}`; const response = await fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: { "Content-Type": "application/json", "x-api-key": apiKey, "anthropic-version": "2023-06-01", }, body: JSON.stringify({ model: "claude-sonnet-4-20250514", max_tokens: 400, system: systemPrompt, messages: [{ role: "user", content: userMessage }], }), }); const data = await response.json(); if (data.error) { return new Response(JSON.stringify({ error: data.error.message }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } const answer = data.content?.[0]?.text; if (!answer) { return new Response(JSON.stringify({ error: "no response" }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } return new Response(JSON.stringify({ answer }), { headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } catch (err) { return new Response(JSON.stringify({ error: err.message || "unknown error" }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } });
